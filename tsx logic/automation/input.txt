import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'FrictionFluidite',
  durationInSeconds: 30,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const FrictionFluidite: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Slow camera drift
  const camX = Math.sin(frame * 0.01) * 14;
  const camScale = interpolate(frame, [0, durationInFrames], [1.05, 1.12], { extrapolateRight: 'clamp' });

  // Divider position between friction (left) and fluidity (right)
  const split = sp(20, { damping: 14 });

  return (
    <AbsoluteFill style={{ backgroundColor: '#504099', overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill
        style={{
          background: 'radial-gradient(circle at 30% 30%, #6B5BB5 0%, #504099 45%, #3B2F73 100%)',
          transform: `scale(${camScale}) translateX(${camX}px)`,
        }}
      />

      {/* Punk collage torn-paper shapes */}
      {[...Array(6)].map((_, i) => {
        const r = seededRandom(i + 3);
        const rot = interpolate(frame, [0, durationInFrames], [r * 20 - 10, r * 20 + 6]);
        return (
          <div
            key={`collage-${i}`}
            style={{
              position: 'absolute',
              left: `${r * 80}%`,
              top: `${seededRandom(i + 9) * 80}%`,
              width: 220 + r * 160,
              height: 90 + r * 60,
              background: i % 2 === 0 ? 'rgba(0,0,0,0.10)' : 'rgba(245,245,245,0.06)',
              transform: `rotate(${rot}deg)`,
              clipPath: 'polygon(0 8%, 96% 0, 100% 92%, 4% 100%)',
            }}
          />
        );
      })}

      {/* Motion texture: halftone dot grid */}
      {[...Array(70)].map((_, i) => {
        const gx = (i % 10) * 190 + 60;
        const gy = Math.floor(i / 10) * 155 + 60;
        const puls = 0.5 + Math.sin(frame * 0.05 + i) * 0.5;
        return (
          <div
            key={`dot-${i}`}
            style={{
              position: 'absolute', left: gx, top: gy,
              width: 6, height: 6, borderRadius: '50%',
              background: '#F5F5F5', opacity: 0.05 + puls * 0.05,
            }}
          />
        );
      })}

      {/* Animated shape layers (slow drifting geometry) */}
      <div style={{
        position: 'absolute', left: '10%', top: '20%', width: 500, height: 500,
        border: '2px solid rgba(245,245,245,0.06)', borderRadius: '50%',
        transform: `scale(${1 + Math.sin(frame * 0.03) * 0.05})`,
      }} />

      {/* Floating particles */}
      {[...Array(28)].map((_, i) => {
        const r = seededRandom(i + 50);
        const y = (frame * (0.4 + r) + r * 1080) % 1080;
        return (
          <div key={`p-${i}`} style={{
            position: 'absolute', left: `${r * 100}%`, top: y,
            width: 3 + r * 4, height: 3 + r * 4, borderRadius: '50%',
            background: i % 3 === 0 ? '#FFC107' : '#F5F5F5', opacity: 0.25,
          }} />
        );
      })}

      {/* Center divider */}
      <div style={{
        position: 'absolute', left: '50%', top: 0, bottom: 0, width: 4,
        background: 'rgba(0,0,0,0.35)', transform: `scaleY(${split})`, transformOrigin: 'center',
      }} />

      {/* LEFT: FRICTION for spending — rough gear/brake */}
      <div style={{
        position: 'absolute', left: '9%', top: '30%',
        transform: `translateY(${(1 - sp(30)) * 60}px)`, opacity: sp(30), textAlign: 'center',
      }}>
        <svg width="230" height="230" viewBox="0 0 100 100">
          <g transform={`rotate(${Math.sin(frame * 0.08) * 8} 50 50)`}>
            {[...Array(10)].map((_, i) => (
              <rect key={i} x="47" y="4" width="6" height="14" fill="#000000"
                transform={`rotate(${i * 36} 50 50)`} />
            ))}
            <circle cx="50" cy="50" r="30" fill="none" stroke="#000000" strokeWidth="8" />
            <circle cx="50" cy="50" r="10" fill="#000000" />
          </g>
        </svg>
        {/* friction sparks */}
        {[...Array(5)].map((_, i) => {
          const s = (frame * 4 + i * 20) % 100;
          return <div key={i} style={{
            position: 'absolute', left: 180 + i * 6, top: 40 + s,
            width: 4, height: 4, background: '#FFC107', opacity: 1 - s / 100,
          }} />;
        })}
        <div style={{ fontFamily: 'Anton', fontSize: 76, color: '#F5F5F5', marginTop: 20, letterSpacing: 1 }}>
          FRICTION
        </div>
        <div style={{ fontFamily: 'Montserrat', fontSize: 34, color: '#000000', fontWeight: 700 }}>
          pour dépenser
        </div>
      </div>

      {/* RIGHT: FLUIDITY for saving — flowing arrow */}
      <div style={{
        position: 'absolute', right: '9%', top: '30%',
        transform: `translateY(${(1 - sp(45)) * 60}px)`, opacity: sp(45), textAlign: 'center',
      }}>
        <svg width="230" height="230" viewBox="0 0 100 100">
          {[...Array(3)].map((_, i) => {
            const flow = (frame * 2 + i * 33) % 100;
            return (
              <path key={i} d="M10,50 Q35,20 60,50 T90,50" fill="none"
                stroke="#FFC107" strokeWidth="5" strokeLinecap="round"
                strokeDasharray="14 12" strokeDashoffset={-flow} opacity={0.4 + i * 0.2} />
            );
          })}
          <polygon points="82,42 96,50 82,58" fill="#F5F5F5" />
        </svg>
        <div style={{ fontFamily: 'Anton', fontSize: 76, color: '#FFC107', marginTop: 20, letterSpacing: 1 }}>
          FLUIDITÉ
        </div>
        <div style={{ fontFamily: 'Montserrat', fontSize: 34, color: '#F5F5F5', fontWeight: 700 }}>
          pour épargner
        </div>
      </div>

      {/* Bottom key line */}
      <div style={{
        position: 'absolute', bottom: 90, width: '100%', textAlign: 'center',
        opacity: sp(70), fontFamily: 'Bricolage Grotesque', fontSize: 40, color: '#F5F5F5',
      }}>
        Le chemin de <span style={{ color: '#FFC107' }}>moindre résistance</span> décide de tout.
      </div>

      {/* Vignette overlay */}
      <AbsoluteFill style={{ boxShadow: 'inset 0 0 320px rgba(0,0,0,0.55)', pointerEvents: 'none' }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45, fontFamily: 'Montserrat',
        fontSize: 22, fontWeight: 600, color: '#9E9E9E', opacity: 0.35,
      }}>
        Matures Finance
      </div>
    </AbsoluteFill>
  );
};

export default FrictionFluidite;
