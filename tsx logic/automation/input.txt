  fps: 30,
  width: 1920,
  height: 1080,

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const { fps, durationInFrames } = useVideoConfig();

    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',

    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.quad),

  const sliceTilt = interpolate(frame, [150, 200], [0, 9], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' });

  // Screen shake at slice impact

  const shakeX = Math.sin(frame * 2.3) * shakeAmp;
  const shakeY = Math.cos(frame * 1.9) * shakeAmp * 0.6;

  const stampIn = sp(230, { damping: 9, stiffness: 200 });

  return (
    <AbsoluteFill style={{ backgroundColor: '#76ABAE', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient + Shape Layers + Texture + Particles + Content + Vignette + Watermark */}

      }} />

        <div key={i} style={{
          position: 'absolute', left: 0, right: 0, top: 120 * i + ((frame * 0.4) % 120),
          height: 1, background: 'rgba(63,79,68,0.07)',
        }} />
      ))}

      {/* Layer 4: Floating particles */}
