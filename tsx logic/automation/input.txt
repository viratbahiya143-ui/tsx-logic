import React from 'react';

import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01MorningChai',
  durationInSeconds: 30,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01MorningChai: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Slow camera zoom-in
  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.08], { extrapolateRight: 'clamp' });

  // Word-by-word kinetic reveal
  const words = [
    { text: 'सुबह की', delay: 15, size: 76 },
    { text: 'पहली चाय', delay: 40, size: 150 },
    { text: 'दूध', delay: 150, size: 190 },
    { text: 'हल्दी की एक चुटकी', delay: 320, size: 110 },
  ];

  // Steam lines rising (micro animation)
  const steam = [0, 1, 2].map((i) => {
    const rise = ((frame * (1.2 + i * 0.3)) % 200);
    return { x: 880 + i * 80, y: 760 - rise, op: interpolate(rise, [0, 100, 200], [0, 0.35, 0]) };
  });

  return (
    <AbsoluteFill style={{ backgroundColor: '#8CA9FF', overflow: 'hidden' }}>
      <AbsoluteFill style={{ opacity: exitOp, transform: `scale(${camZoom})` }}>
        {/* Gradient */}
        <AbsoluteFill style={{ background: 'radial-gradient(circle at 50% 30%, #A9BFFF 0%, #8CA9FF 55%, #6E8DE8 100%)' }} />

        {/* Shape Layers */}
        <div style={{
          position: 'absolute', width: 700, height: 700, borderRadius: '50%',
          border: '3px solid rgba(238,238,238,0.15)', left: -200, top: -250,
          transform: `rotate(${frame * 0.1}deg) scale(${1 + Math.sin(frame * 0.02) * 0.03})`,
        }} />
        <div style={{
          position: 'absolute', width: 500, height: 500, background: 'rgba(0,0,0,0.04)',
          right: -150, bottom: -180, transform: `rotate(${45 + frame * 0.08}deg)`,
        }} />

        {/* Texture: thin horizontal lines */}
        {[...Array(6)].map((_, i) => (
          <div key={i} style={{
            position: 'absolute', left: 0, right: 0, top: 180 * (i + 1),
            height: 1, background: 'rgba(0,0,0,0.05)',
            transform: `translateX(${Math.sin(frame * 0.01 + i) * 20}px)`,
          }} />
        ))}

        {/* Particles */}
        {[...Array(14)].map((_, i) => {
          const px = seededRandom(i * 3) * 1920;
          const py = (seededRandom(i * 7) * 1080 + frame * (0.3 + seededRandom(i) * 0.5)) % 1080;
          return (
            <div key={i} style={{
              position: 'absolute', left: px, top: py, width: 5 + seededRandom(i * 5) * 5,
              height: 5 + seededRandom(i * 5) * 5, borderRadius: '50%',
              background: 'rgba(238,238,238,0.35)',
            }} />
          );
        })}

        {/* Steam micro-animation */}
        {steam.map((s, i) => (
          <div key={i} style={{
            position: 'absolute', left: s.x, top: s.y, width: 4, height: 70,
            borderRadius: 4, background: `rgba(238,238,238,${s.op})`,
            transform: `rotate(${Math.sin(frame * 0.08 + i) * 8}deg)`,
          }} />
        ))}

        {/* Content: kinetic typography stack */}
        <AbsoluteFill style={{ justifyContent: 'center', alignItems: 'center', flexDirection: 'column', gap: 18 }}>
          {words.map((w, i) => {
            const s = sp(w.delay);
            const active = i === 2; // दूध gets highlight
            return (
              <div key={i} style={{
                fontFamily: "'Yantra One', 'Noto Sans Devanagari', sans-serif",
                fontSize: w.size, fontWeight: 800, lineHeight: 1.05,
                color: active ? '#EEEEEE' : '#000000',
                background: active ? '#000000' : 'transparent',
                padding: active ? '0 40px' : 0, borderRadius: active ? 20 : 0,
                opacity: Math.min(1, s), transform: `translateY(${(1 - s) * 80}px) scale(${0.9 + s * 0.1})`,
              }}>
                {w.text}
              </div>
            );
          })}
        </AbsoluteFill>

        {/* Vignette */}
        <AbsoluteFill style={{ background: 'radial-gradient(circle, transparent 55%, rgba(0,0,0,0.28) 100%)', pointerEvents: 'none' }} />

        {/* Watermark */}
        <div style={{
          position: 'absolute', bottom: 30, right: 45, fontFamily: 'Inter, sans-serif',
          fontSize: 22, fontWeight: 600, color: '#000000',
          opacity: interpolate(frame, [0, 60], [0.6, 0.3], { extrapolateRight: 'clamp' }),
        }}>
          ProBuffet HQ
        </div>
      </AbsoluteFill>
    </AbsoluteFill>
  );
};

export default Scene01MorningChai;
