import {
  useCurrentFrame,
  useVideoConfig,
  interpolate,
  Easing,
  spring,
  AbsoluteFill,
} from 'remotion';

export const compositionConfig = {
  id: 'Scene01',
  durationInSeconds: 17,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({
      frame: Math.max(0, frame - delay),
      fps,
      config: { damping: 12, mass: 0.4, stiffness: 120, ...config },
    });

  const exitOp = interpolate(
    frame,
    [durationInFrames - 15, durationInFrames],
    [1, 0],
    { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  // Question mark pulse
  const qPulse = 1 + Math.sin(frame * 0.08) * 0.04;

  // Salary bar fill animation (starts at frame 30)
  const barFill = interpolate(frame, [30, 90], [0, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });

  // Cut indicator drop
  const cutDrop = sp(80, { damping: 10, stiffness: 90 });

  // Headline reveal
  const headlineOp = interpolate(frame, [0, 20], [0, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });
  const headlineY = interpolate(frame, [0, 20], [40, 0], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });

  // Sub text
  const subOp = interpolate(frame, [25, 45], [0, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  // Floating particles
  const particles = Array.from({ length: 18 }, (_, i) => ({
    x: seededRandom(i * 7) * 1920,
    y: seededRandom(i * 13) * 1080,
    r: seededRandom(i * 3) * 6 + 3,
    speed: seededRandom(i * 17) * 0.5 + 0.2,
    phase: seededRandom(i * 5) * Math.PI * 2,
  }));

  // Screen shake at frame 60
  const shakeX =
    frame >= 58 && frame <= 65
      ? Math.sin(frame * 15) * 5 * (1 - (frame - 58) / 7)
      : 0;

  return (
    <AbsoluteFill
      style={{
        backgroundColor: '#00ADB5',
        overflow: 'hidden',
        opacity: exitOp,
        transform: `translateX(${shakeX}px)`,
      }}
    >
      {/* === GRADIENT OVERLAY === */}
      <AbsoluteFill
        style={{
          background:
            'radial-gradient(ellipse at 20% 50%, rgba(13,115,119,0.55) 0%, transparent 65%), radial-gradient(ellipse at 85% 20%, rgba(34,40,49,0.4) 0%, transparent 60%)',
        }}
      />

      {/* === GEOMETRIC SHAPE LAYERS === */}
      <AbsoluteFill style={{ pointerEvents: 'none' }}>
        {/* Large rotating circle BG */}
        <div
          style={{
            position: 'absolute',
            top: -200,
            right: -150,
            width: 700,
            height: 700,
            borderRadius: '50%',
            border: '2px solid rgba(238,238,238,0.06)',
            transform: `rotate(${frame * 0.3}deg) scale(${1 + Math.sin(frame * 0.02) * 0.03})`,
          }}
        />
        <div
          style={{
            position: 'absolute',
            top: -100,
            right: -50,
            width: 500,
            height: 500,
            borderRadius: '50%',
            border: '1.5px solid rgba(238,238,238,0.04)',
            transform: `rotate(${-frame * 0.2}deg)`,
          }}
        />
        {/* Triangle SVG shape */}
        <svg
          style={{
            position: 'absolute',
            bottom: 60,
            left: -60,
            opacity: 0.05,
            transform: `rotate(${frame * 0.15}deg)`,
          }}
          width="400"
          height="400"
          viewBox="0 0 400 400"
        >
          <polygon points="200,20 380,380 20,380" fill="none" stroke="#EEEEEE" strokeWidth="2" />
        </svg>
        {/* Rect */}
        <div
          style={{
            position: 'absolute',
            bottom: 200,
            right: 80,
            width: 220,
            height: 220,
            border: '1.5px solid rgba(238,238,238,0.05)',
            transform: `rotate(${25 + frame * 0.1}deg)`,
          }}
        />
      </AbsoluteFill>

      {/* === MOTION TEXTURE — diagonal lines === */}
      <AbsoluteFill style={{ pointerEvents: 'none', opacity: 0.06 }}>
        <svg width="1920" height="1080" viewBox="0 0 1920 1080">
          {Array.from({ length: 30 }, (_, i) => (
            <line
              key={i}
              x1={i * 70 - 200}
              y1={0}
              x2={i * 70 + 400}
              y2={1080}
              stroke="#EEEEEE"
              strokeWidth="1"
            />
          ))}
        </svg>
      </AbsoluteFill>

      {/* === FLOATING PARTICLES === */}
      <AbsoluteFill style={{ pointerEvents: 'none' }}>
        {particles.map((p, i) => (
          <div
            key={i}
            style={{
              position: 'absolute',
              left: p.x + Math.sin(frame * p.speed + p.phase) * 12,
              top: p.y + Math.cos(frame * p.speed * 0.7 + p.phase) * 8,
              width: p.r,
              height: p.r,
              borderRadius: '50%',
              backgroundColor: i % 3 === 0 ? '#F2B705' : '#EEEEEE',
              opacity: 0.18 + Math.sin(frame * 0.05 + p.phase) * 0.08,
            }}
          />
        ))}
      </AbsoluteFill>

      {/* === MAIN CONTENT === */}
      <AbsoluteFill
        style={{
          display: 'flex',
          flexDirection: 'column',
          justifyContent: 'center',
          paddingLeft: 120,
          paddingRight: 900,
        }}
      >
        {/* QUESTION MARK HERO */}
        <div
          style={{
            transform: `scale(${qPulse})`,
            transformOrigin: 'left center',
            opacity: headlineOp,
            marginBottom: 24,
          }}
        >
          <svg width="90" height="90" viewBox="0 0 90 90">
            <circle cx="45" cy="45" r="43" fill="none" stroke="#F2B705" strokeWidth="3" />
            <text
              x="45"
              y="62"
              textAnchor="middle"
              fontSize="52"
              fontFamily="Inter, sans-serif"
              fontWeight="700"
              fill="#F2B705"
            >
              ?
            </text>
          </svg>
        </div>

        {/* HEADLINE */}
        <div
          style={{
            fontFamily: "'Noto Sans Devanagari', sans-serif",
            fontSize: 102,
            fontWeight: 800,
            color: '#EEEEEE',
            lineHeight: 1.1,
            opacity: headlineOp,
            transform: `translateY(${headlineY}px)`,
            marginBottom: 20,
          }}
        >
          एक सवाल
        </div>
        <div
          style={{
            fontFamily: "'Noto Sans Devanagari', sans-serif",
            fontSize: 59,
            fontWeight: 600,
            color: '#222831',
            lineHeight: 1.3,
            opacity: subOp,
            maxWidth: 680,
          }}
        >
          जिस दिन सैलरी आती है —<br />
          उससे पहले ही कट चुका होता है।
        </div>
      </AbsoluteFill>

      {/* === SALARY BAR VISUAL (Right side) === */}
      <AbsoluteFill
        style={{
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'flex-end',
          paddingRight: 100,
        }}
      >
        <div
          style={{
            opacity: sp(30),
            transform: `translateX(${interpolate(sp(30), [0, 1], [80, 0])})`,
          }}
        >
          {/* Salary bar container */}
          <div style={{ position: 'relative', width: 260, height: 520 }}>
            {/* Label top */}
            <div
              style={{
                fontFamily: 'Inter, sans-serif',
                fontSize: 28,
                fontWeight: 700,
                color: '#EEEEEE',
                marginBottom: 14,
                letterSpacing: 2,
                textTransform: 'uppercase',
              }}
            >
              SALARY
            </div>

            {/* Full bar */}
            <div
              style={{
                width: 260,
                height: 420,

                position: 'relative',
                overflow: 'hidden',

              }}
            >
              {/* Green fill = what you keep */}
              <div
                style={{

              <div

                  backgroundColor: '#FF6B6B',
                  borderRadius: '14px 14px 0 0',
                  display: 'flex',
                  alignItems: 'center',
                  justifyContent: 'center',
                  flexDirection: 'column',
                }}
              >
                {barFill > 0.5 && (
                  <>
                    {/* Scissors SVG icon */}
                    <svg width="38" height="38" viewBox="0 0 24 24" fill="none">
                      <circle cx="6" cy="6" r="3" stroke="#EEEEEE" strokeWidth="1.5" />
                      <circle cx="6" cy="18" r="3" stroke="#EEEEEE" strokeWidth="1.5" />
                      <line x1="8.5" y1="8.5" x2="21" y2="3" stroke="#EEEEEE" strokeWidth="1.5" strokeLinecap="round" />

                      CUT

              </div>

              {/* Divider line with animation */}
              <div
                style={{
                  position: 'absolute',
                  top: `${barFill * 40}%`,
                  width: '100%',
                  height: 3,
                  backgroundColor: '#F2B705',
                  opacity: cutDrop,
                }}
              />
            </div>

            {/* Labels */}
            <div style={{ display: 'flex', justifyContent: 'space-between', marginTop: 10 }}>

                60% बचता

      {/* === VIGNETTE === */}
      <AbsoluteFill
        style={{
          background:
            'radial-gradient(ellipse at center, transparent 55%, rgba(13,40,49,0.55) 100%)',
          pointerEvents: 'none',

      />

      {/* === WATERMARK === */}
      <div
        style={{
          position: 'absolute',
          bottom: 30,
          right: 45,
          fontFamily: 'Inter, sans-serif',
          fontSize: 22,
