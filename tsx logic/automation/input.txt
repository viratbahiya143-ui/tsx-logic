  const gradAngle = interpolate(frame, [0, durationInFrames], [135, 165]);

  // --- FLOATING GEOMETRIC SHAPES ---
  const shapes = Array.from({ length: 14 }, (_, i) => {
    const s = seededRandom(i + 1);
    const size = 60 + s * 200;
    const startX = seededRandom(i + 50) * 1920;
    const startY = seededRandom(i + 100) * 1080;
    const drift = Math.sin(frame * 0.008 + i) * 30;
    const rot = Math.sin(frame * 0.005 + i * 2) * 15;
    const opacity = 0.03 + seededRandom(i + 200) * 0.05;
    const shapeType = i % 3;
    return (
      <div
        key={`shape-${i}`}
        style={{
          position: 'absolute',
          left: startX + drift,

          height: size,

          border: `2px solid rgba(240,240,240,${opacity})`,

          background: shapeType === 0 ? `radial-gradient(circle, rgba(240,240,240,${opacity * 0.5}) 0%, transparent 70%)` : 'transparent',
        }}
      />
    );
  });

  // --- GRID TEXTURE ---
  const gridLines = Array.from({ length: 20 }, (_, i) => {
    const isHorizontal = i < 10;
    const pos = (i % 10) * 108 + 54;
    const gridOp = interpolate(frame, [0, 60], [0, 0.06], { extrapolateRight: 'clamp' });
    return (
      <div
        key={`grid-${i}`}
        style={{
          position: 'absolute',
          left: isHorizontal ? 0 : pos,
          top: isHorizontal ? pos : 0,
          width: isHorizontal ? '100%' : 1,
          height: isHorizontal ? 1 : '100%',
          backgroundColor: `rgba(240,240,240,${gridOp})`,
        }}
      />
    );
  });

  // --- PARTICLES ---
  const particles = Array.from({ length: 30 }, (_, i) => {
    const px = seededRandom(i + 300) * 1920;
    const py = seededRandom(i + 400) * 1080;
    const pSize = 2 + seededRandom(i + 500) * 4;
    const pOp = 0.15 + Math.sin(frame * 0.04 + i * 0.7) * 0.1;
    const pDrift = Math.sin(frame * 0.02 + i) * 15;
    return (
      <div
        key={`p-${i}`}
        style={{
          position: 'absolute',
          left: px + pDrift,
          top: py + Math.cos(frame * 0.015 + i) * 12,
          width: pSize,
          height: pSize,
          borderRadius: '50%',
          backgroundColor: `rgba(240,240,240,${pOp})`,
        }}
      />
    );
  });

  // --- POPULATION DOT GRID (88 million visual) ---
  const dotGridEntrance = sp(20, { damping: 14, stiffness: 80 });
  const totalDots = 100; // 10x10 grid representing US population
  const filledDots = 27; // ~27% representing 88M of ~330M
  const dotSize = 14;
  const dotGap = 22;
  const gridW = 10 * (dotSize + dotGap);
  const gridH = 10 * (dotSize + dotGap);

  const dotGrid = Array.from({ length: totalDots }, (_, i) => {
    const row = Math.floor(i / 10);
    const col = i % 10;
    const dotDelay = 25 + i * 0.8;
    const dotSp = sp(dotDelay, { damping: 10, stiffness: 140 });
    const isFilled = i < filledDots;
    const pulseScale = isFilled ? 1 + Math.sin(frame * 0.06 + i * 0.3) * 0.08 : 1;

    // Animate fill transition
    const fillPhase = interpolate(frame, [80, 80 + i * 1.5], [0, 1], {
      extrapolateLeft: 'clamp',
      extrapolateRight: 'clamp',
    });

    return (
      <div
        key={`dot-${i}`}
        style={{
          position: 'absolute',
          left: col * (dotSize + dotGap),
          top: row * (dotSize + dotGap),

          height: dotSize,
          borderRadius: '50%',

          transition: 'background-color 0.3s',
        }}
      />
    );
  });

  // --- BIG NUMBER "88M" ---
  const numEntrance = sp(15, { damping: 10, stiffness: 100 });
  const numSlide = interpolate(numEntrance, [0, 1], [80, 0]);
  const numCountUp = interpolate(frame, [15, 75], [0, 88], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',

  const pctSlide = interpolate(pctEntrance, [0, 1], [60, 0]);
  const pctCount = interpolate(frame, [90, 150], [0, 54], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  // --- SUBTITLE TEXT ---
  const subEntrance = sp(50, { damping: 14 });
  const sub2Entrance = sp(130, { damping: 14 });

  // --- ALERT PULSE LINE ---
  const pulseWidth = interpolate(frame, [0, durationInFrames], [0, 1920 * 3], {
    extrapolateRight: 'clamp',
  });
  const pulseOp = 0.12 + Math.sin(frame * 0.08) * 0.06;

  // --- DANGER BAR (bottom) ---
  const dangerWidth = interpolate(frame, [60, 200], [0, 100], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  // --- CAMERA ZOOM ---
  const camZoom = interpolate(frame, [0, durationInFrames], [1.04, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });
  const camX = interpolate(frame, [0, durationInFrames], [-10, 5]);
  const camY = interpolate(frame, [0, durationInFrames], [-5, 3]);

  // --- SCREEN SHAKE at impact ---

  const shakeY = frame > 70 && frame < 85 ? Math.cos(frame * 2) * 2 : 0;

  // --- VIGNETTE ---
  const vignette = `radial-gradient(ellipse at center, transparent 50%, rgba(0,0,0,0.45) 100%)`;

  return (
    <AbsoluteFill
      style={{
        background: bgGrad,
        overflow: 'hidden',
        opacity: exitOp,

      {/* Camera wrapper */}
      <div
        style={{
          position: 'absolute',
          inset: 0,
          transform: `scale(${camZoom}) translate(${camX + shakeX}px, ${camY + shakeY}px)`,
        }}
      >
        {/* Grid texture */}
        {gridLines}

        {/* Geometric shapes */}
        {shapes}

        {/* Particles */}
        {particles}

        {/* Horizontal pulse line */}
        <div
          style={{
            position: 'absolute',
            top: 540,
            left: -100,

            height: 2,
