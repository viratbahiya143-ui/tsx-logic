import React from 'react';
import {
  AbsoluteFill,
  Easing,
  interpolate,
  spring,
  useCurrentFrame,
  useVideoConfig,
} from 'remotion';

export const compositionConfig = {
  id: 'Scene01MoneyLeaving',
  durationInFrames: 500, // 16.67 seconds × 30 FPS
  fps: 30,
  width: 1920,
  height: 1080,
};

const BG = '#FFEA11';
const INK = '#121212';
const PAPER = '#F5F5F5';
const RED = '#FF2E2E';

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

type SpringConfig = Partial<{
  damping: number;
  mass: number;
  stiffness: number;
  overshootClamping: boolean;
}>;

type CardData = {
  label: string;
  sub: string;
  delay: number;
  icon: 'bank' | 'chart' | 'house';
};

const cards: CardData[] = [
  {
    label: 'FD',
    sub: 'अब वहाँ नहीं जा रहा',
    delay: 22,
    icon: 'bank',
  },
  {
    label: 'शेयर बाज़ार',
    sub: 'डर के निकल रहा है',
    delay: 52,
    icon: 'chart',
  },
  {
    label: 'प्रॉपर्टी',
    sub: 'EMI भारी हो गई',
    delay: 82,
    icon: 'house',
  },
];

const renderIcon = (type: CardData['icon']) => {
  if (type === 'bank') {
    return (
      <svg width="120" height="120" viewBox="0 0 100 100">
        <polygon
          points="50,8 92,32 8,32"
          fill="none"
          stroke={INK}
          strokeWidth="6"
          strokeLinejoin="round"
        />

        <rect x="16" y="40" width="10" height="34" fill={INK} />
        <rect x="45" y="40" width="10" height="34" fill={INK} />
        <rect x="74" y="40" width="10" height="34" fill={INK} />
        <rect x="8" y="80" width="84" height="10" fill={INK} />
      </svg>
    );
  }

  if (type === 'chart') {
    return (
      <svg width="120" height="120" viewBox="0 0 100 100">
        <polyline
          points="8,20 34,48 52,36 90,82"
          fill="none"
          stroke={INK}
          strokeWidth="7"
          strokeLinecap="round"
          strokeLinejoin="round"
        />

        <polygon points="90,82 70,76 84,62" fill={INK} />
      </svg>
    );
  }

  return (
    <svg width="120" height="120" viewBox="0 0 100 100">
      <polygon
        points="50,10 92,46 82,46 82,88 18,88 18,46 8,46"
        fill="none"
        stroke={INK}
        strokeWidth="6"
        strokeLinejoin="round"
      />

      <rect x="42" y="60" width="16" height="28" fill={INK} />
    </svg>
  );
};

export const Scene01MoneyLeaving: React.FC = () => {
  const frame = useCurrentFrame();
  const {fps, durationInFrames} = useVideoConfig();

  const sp = (delay: number, config: SpringConfig = {}) => {
    return spring({
      frame: Math.max(0, frame - delay),
      fps,
      config: {
        damping: 12,
        mass: 0.4,
        stiffness: 120,
        ...config,
      },
    });
  };

  // Scene fade-out
  const exitOpacity = interpolate(
    frame,
    [durationInFrames - 15, durationInFrames],
    [1, 0],
    {
      extrapolateLeft: 'clamp',
      extrapolateRight: 'clamp',
    },
  );

  // Camera animation
  const cameraScale = interpolate(
    frame,
    [0, durationInFrames],
    [1, 1.09],
    {
      extrapolateLeft: 'clamp',
      extrapolateRight: 'clamp',
    },
  );

  const cameraX = interpolate(
    frame,
    [0, durationInFrames],
    [0, -14],
    {
      extrapolateLeft: 'clamp',
      extrapolateRight: 'clamp',
    },
  );

  // Screen shake on red X stamp impacts
  const stampTimes = [60, 90, 120];

  const shake = stampTimes.reduce((total, stampFrame) => {
    const elapsed = frame - stampFrame;

    if (elapsed >= 0 && elapsed <= 9) {
      return total + Math.sin(elapsed * 2.4) * (9 - elapsed);
    }

    return total;
  }, 0);

  const kickerOpacity = interpolate(frame, [130, 150], [0, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  const kickerTranslateY = interpolate(frame, [130, 150], [40, 0], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });

  return (
    <AbsoluteFill
      style={{
        backgroundColor: BG,
        overflow: 'hidden',
      }}
    >
      <AbsoluteFill
        style={{
          opacity: exitOpacity,
          transform: `scale(${cameraScale}) translate(${
            cameraX + shake
          }px, ${shake * 0.6}px)`,
        }}
      >
        {/* Dynamic gradient */}
        <AbsoluteFill
          style={{
            background: `radial-gradient(
              circle at ${30 + Math.sin(frame * 0.01) * 10}% 20%,
              rgba(245, 245, 245, 0.35),
              transparent 60%
            )`,
          }}
        />

        {/* Paper strips */}
        {[0, 1, 2, 3].map((i) => (
          <div
            key={`strip-${i}`}
            style={{
              position: 'absolute',
              left: seededRandom(i + 1) * 1600 - 100,
              top: seededRandom(i + 7) * 900 - 60,
              width: 560,
              height: 120,
              background:
                i % 2 === 0
                  ? 'rgba(18, 18, 18, 0.06)'
                  : 'rgba(245, 245, 245, 0.4)',
              border: '3px dashed rgba(18, 18, 18, 0.12)',
              transform: `rotate(${
                -14 + i * 9 + Math.sin(frame * 0.008 + i) * 2
              }deg) translateY(${
                Math.sin(frame * 0.015 + i * 2) * 10
              }px)`,
            }}
          />
        ))}

        {/* Halftone texture */}
        <AbsoluteFill
          style={{
            backgroundImage: `radial-gradient(${INK} 1.6px, transparent 1.6px)`,
            backgroundSize: '36px 36px',
            opacity: 0.07,
            transform: `translateX(${(frame * 0.3) % 36}px)`,
          }}
        />

        {/* Floating particles */}
        {Array.from({length: 14}).map((_, i) => {
          const particleSize = 5 + seededRandom(i + 60) * 7;
          const particleSpeed = 0.3 + seededRandom(i) * 0.5;

          return (
            <div
              key={`particle-${i}`}
              style={{
                position: 'absolute',
                left: seededRandom(i + 20) * 1920,
                top:
                  (seededRandom(i + 40) * 1080 +
                    frame * particleSpeed) %
                  1080,
                width: particleSize,
                height: particleSize,
                borderRadius: '50%',
                backgroundColor: INK,
                opacity: 0.14,
              }}
            />
          );
        })}

        {/* Heading */}
        <div
          style={{
            position: 'absolute',
            top: 130,
            width: '100%',
            textAlign: 'center',
          }}
        >
          <div
            style={{
              display: 'inline-block',
              fontFamily:
                'Yantra One, Noto Sans Devanagari, sans-serif',

          >
            पैसा निकल रहा है
