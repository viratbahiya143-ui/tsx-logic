import React from 'react';

import {
  AbsoluteFill,
  Easing,
  interpolate,
  spring,
  useCurrentFrame,
  useVideoConfig,
} from 'remotion';

export const compositionConfig = {
  id: 'Scene01MissingChapter',
  durationInSeconds: 13,
  fps: 30,
  width: 1920,
  height: 1080,
};

const PALETTE = {
  bg: '#CBD2A4',
  paper: '#F4F2F2',
  ink: '#000000',
  moss: '#A3AC72',
  olive: '#7C8449',
  rust: '#C1440E',
  mustard: '#E3B505',
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

// Large transparent geometric shapes drifting slowly behind the content
const S01Shapes: React.FC<{ frame: number }> = ({ frame }) => (
  <AbsoluteFill>
    <svg width="1920" height="1080" viewBox="0 0 1920 1080">
      <circle
        cx={340 + Math.sin(frame * 0.006) * 40}
        cy={300 + Math.cos(frame * 0.005) * 30}
        r={380 + Math.sin(frame * 0.02) * 12}
        fill={PALETTE.ink}
        opacity={0.05}
      />
      <rect
        x={1280}
        y={520}
        width={620}
        height={620}
        fill={PALETTE.olive}
        opacity={0.07}
        transform={`rotate(${12 + Math.sin(frame * 0.008) * 4} 1590 830)`}
      />
      <polygon
        points="960,60 1220,520 700,520"
        fill={PALETTE.rust}
        opacity={0.05}
        transform={`translate(0 ${Math.sin(frame * 0.01) * 18})`}
      />
    </svg>
  </AbsoluteFill>
);

// Motion texture — scrolling notebook rules
const S01Texture: React.FC<{ frame: number }> = ({ frame }) => (
  <AbsoluteFill style={{ opacity: 0.16, mixBlendMode: 'multiply' }}>
    <svg width="1920" height="1080">
      <defs>
        <pattern id="s01rule" width="1920" height="72" patternUnits="userSpaceOnUse">
          <line x1="0" y1="71" x2="1920" y2="71" stroke={PALETTE.ink} strokeWidth="1.5" />
        </pattern>
      </defs>
      <rect
        width="1920"
        height="1160"
        y={-((frame * 0.35) % 72)}
        fill="url(#s01rule)"
      />
    </svg>
  </AbsoluteFill>
);

// Floating particles — torn paper confetti
const S01Particles: React.FC<{ frame: number }> = ({ frame }) => (
  <AbsoluteFill>
    {new Array(22).fill(0).map((_, i) => {
      const sx = seededRandom(i + 3) * 1920;
      const sy = (seededRandom(i + 91) * 1080 + frame * (0.5 + seededRandom(i) * 0.8)) % 1120;
      const size = 6 + seededRandom(i + 44) * 12;
      return (
        <div
          key={i}
          style={{
            position: 'absolute',
            left: sx,
            top: sy - 40,
            width: size,
            height: size * 0.6,
            backgroundColor: i % 3 === 0 ? PALETTE.paper : PALETTE.ink,
            opacity: 0.22,
            transform: `rotate(${frame * (0.4 + seededRandom(i + 7)) + i * 30}deg)`,
          }}
        />
      );
    })}
  </AbsoluteFill>
);

const S01Vignette: React.FC = () => (
  <AbsoluteFill
    style={{
      background:
        'radial-gradient(circle at 50% 45%, rgba(0,0,0,0) 45%, rgba(0,0,0,0.42) 100%)',
    }}
  />
);

const S01Watermark: React.FC<{ frame: number }> = ({ frame }) => {
  const op = interpolate(frame, [10, 30], [0, 0.3], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });
  return (
    <div
      style={{
        position: 'absolute',
        bottom: 30,
        right: 45,
        fontFamily: 'Inter, Poppins, sans-serif',
        fontSize: 22,
        fontWeight: 600,
        color: PALETTE.ink,
        opacity: op,
        letterSpacing: 1,
      }}
    >
      ProBuffet HQ
    </div>
  );
};

const Scene01MissingChapter: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({
      frame: Math.max(0, frame - delay),
      fps,
      config: { damping: 12, mass: 0.4, stiffness: 120, ...config },
    });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });

  // Camera — slow push-in with a punk jitter on impact
  const camZoom = interpolate(frame, [0, durationInFrames], [1.04, 1.12]);
  const shake = frame > 200 && frame < 214 ? Math.sin(frame * 3.4) * 6 : 0;

  const card = sp(6);
  const formulaIn = (i: number) => sp(24 + i * 14);
  const stampScale = interpolate(sp(200, { damping: 9, stiffness: 180 }), [0, 1], [2.6, 1]);
  const stampOp = interpolate(frame, [200, 210], [0, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
  });
  const strike = interpolate(frame, [150, 185], [0, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });

  const formulas = [
    { label: 'sin θ', icon: 'wave' },
    { label: 'cos θ', icon: 'triangle' },
    { label: 'tan θ', icon: 'angle' },
  ];

  return (
    <AbsoluteFill style={{ backgroundColor: PALETTE.bg, overflow: 'hidden', opacity: exitOp }}>
      {/* Dynamic gradient background */}
      <AbsoluteFill
        style={{
          background: `linear-gradient(${140 + Math.sin(frame * 0.01) * 10}deg, #D8DEB4 0%, ${PALETTE.bg} 45%, #B7BF8E 100%)`,
        }}
      />

      <S01Shapes frame={frame} />
      <S01Texture frame={frame} />
      <S01Particles frame={frame} />

      {/* Main content */}
      <AbsoluteFill
        style={{
          transform: `scale(${camZoom}) translateX(${shake}px)`,
          alignItems: 'center',
          justifyContent: 'center',
        }}
      >
        {/* Torn notebook page collage card */}
        <div
          style={{
            width: 1420,
            padding: '58px 70px',
            backgroundColor: PALETTE.paper,
            border: `6px solid ${PALETTE.ink}`,
            boxShadow: `26px 26px 0px ${PALETTE.ink}`,
            transform: `rotate(${-1.6 + Math.sin(frame * 0.02) * 0.4}deg) translateY(${interpolate(
              card,
              [0, 1],
              [90, 0]
            )}px)`,
            opacity: card,
          }}
        >
          <div
            style={{
              fontFamily: 'JetBrains Mono, monospace',
              fontSize: 29,
              letterSpacing: 4,
              color: PALETTE.olive,
              marginBottom: 18,
            }}
          >
            CLASS X — SYLLABUS
          </div>

          <div
            style={{
              fontFamily: "'Yantra One', 'Noto Sans Devanagari', sans-serif",
              fontSize: 102,
              lineHeight: 1.1,
              color: PALETTE.ink,
            }}
          >
            त्रिकोणमिति रटा दी गई
          </div>

          {/* Formula chips with staggered spring entrance */}
          <div style={{ display: 'flex', gap: 30, marginTop: 44 }}>
            {formulas.map((f, i) => {
              const p = formulaIn(i);
              const struck = strike > (i + 1) / 4;
              return (
                <div
                  key={f.label}
                  style={{
                    display: 'flex',
                    alignItems: 'center',
                    gap: 18,
                    padding: '20px 30px',
                    border: `4px solid ${PALETTE.ink}`,
                    backgroundColor: i === 1 ? PALETTE.mustard : PALETTE.bg,
                    transform: `scale(${p}) rotate(${(i - 1) * 1.4}deg)`,
                    opacity: p,
                    position: 'relative',
                  }}
                >
                  <svg width="52" height="52" viewBox="0 0 52 52">
                    {f.icon === 'wave' && (
                      <path
                        d="M4 26 Q13 4 22 26 T40 26 T48 26"
                        fill="none"
                        stroke={PALETTE.ink}
                        strokeWidth="4"
                        strokeLinecap="round"
                      />
                    )}
                    {f.icon === 'triangle' && (
                      <polygon
                        points="6,46 46,46 46,10"
                        fill="none"
                        stroke={PALETTE.ink}
                        strokeWidth="4"
                        strokeLinejoin="round"
                      />
                    )}
                    {f.icon === 'angle' && (
                      <>
                        <path d="M6 46 L46 46 M6 46 L40 12" stroke={PALETTE.ink} strokeWidth="4" strokeLinecap="round" />
                        <path d="M18 46 A14 14 0 0 0 14 36" fill="none" stroke={PALETTE.rust} strokeWidth="4" />
                      </>
                    )}
                  </svg>
                  <span
                    style={{
                      fontFamily: 'JetBrains Mono, monospace',
                      fontSize: 44,
                      fontWeight: 700,
                      color: PALETTE.ink,
                    }}
                  >
                    {f.label}
                  </span>
                  {struck ? (
                    <div
                      style={{
                        position: 'absolute',
                        left: 10,
                        right: 10,
                        top: '50%',
                        height: 6,
                        backgroundColor: PALETTE.rust,
                        transform: 'rotate(-4deg)',
                      }}
                    />
                  ) : null}
                </div>
              );
            })}
          </div>
        </div>

        {/* Rubber stamp payoff — EMI chapter missing */}
        <div
          style={{
            position: 'absolute',
            bottom: 118,
            right: 190,
            padding: '26px 44px',
            border: `8px solid ${PALETTE.rust}`,
            color: PALETTE.rust,
            fontFamily: "'Yantra One', 'Noto Sans Devanagari', sans-serif",
            fontSize: 65,
            letterSpacing: 2,
            backgroundColor: 'rgba(244,242,242,0.82)',
            transform: `rotate(-9deg) scale(${stampScale})`,
            opacity: stampOp,
            display: 'flex',
            alignItems: 'center',
            gap: 22,
          }}
        >
          <svg width="60" height="60" viewBox="0 0 60 60">
            <rect x="6" y="10" width="48" height="40" rx="4" fill="none" stroke={PALETTE.rust} strokeWidth="5" />
            <path d="M6 22 H54" stroke={PALETTE.rust} strokeWidth="5" />
            <path d="M18 38 H30" stroke={PALETTE.rust} strokeWidth="5" strokeLinecap="round" />
          </svg>
          EMI — चैप्टर गायब
        </div>
      </AbsoluteFill>

      <S01Vignette />
      <S01Watermark frame={frame} />
    </AbsoluteFill>
  );
};

export default Scene01MissingChapter;
