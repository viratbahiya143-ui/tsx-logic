  id: 'Scene01GoldPriceShock',
  durationInSeconds: 22,
  fps: 30,
  width: 1920,
  height: 1080,
};

  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);

    extrapolateRight: 'clamp',

  const camY = Math.sin(frame * 0.012) * 10;

  // Animated price counter 0 -> 1,44,000
  const priceProgress = interpolate(frame, [25, 115], [0, 1], {
    extrapolateLeft: 'clamp',
    extrapolateRight: 'clamp',
    easing: Easing.out(Easing.cubic),
  });
  const price = Math.round(interpolate(priceProgress, [0, 1], [0, 144000]));

  // Impact shake when the number locks in
  const shake =
    frame > 112 && frame < 130 ? Math.sin((frame - 112) * 2.2) * (130 - frame) * 0.7 : 0;

  const kicker = sp(10);
  const heroIn = sp(20, { damping: 14 });
  const barsIn = sp(140, { damping: 16 });
  const strapIn = sp(210, { damping: 15 });

    <AbsoluteFill style={{ backgroundColor: '#90CAF9', overflow: 'hidden' }}>
      {/* Dynamic gradient background */}

        }}
      />

      {/* Punk collage halftone sheet */}
      <AbsoluteFill
        style={{
          backgroundImage: 'radial-gradient(#000000 2px, transparent 2px)',
          backgroundSize: '26px 26px',
          opacity: 0.13,
          transform: `translateY(${Math.sin(frame * 0.02) * 8}px)`,
        }}
      />

      {/* Animated shape layers — large transparent geometry drifting slowly */}
      <AbsoluteFill>
        <div
          style={{
            position: 'absolute',
            left: -180,
            top: 120,
            width: 760,
            height: 760,
            borderRadius: '50%',
            border: '10px solid #000000',
            opacity: 0.06,
            transform: `rotate(${frame * 0.12}deg) scale(${1 + Math.sin(frame * 0.02) * 0.05})`,
          }}
        />
        <div
          style={{
            position: 'absolute',
            right: -140,
            top: -120,
            width: 640,
            height: 640,
            backgroundColor: '#000000',
            opacity: 0.05,
            transform: `rotate(${18 + Math.sin(frame * 0.015) * 6}deg)`,
          }}
        />
        <div
          style={{
            position: 'absolute',
            left: 620,
            bottom: -220,
            width: 0,
            height: 0,
            borderLeft: '340px solid transparent',
            borderRight: '340px solid transparent',
            borderBottom: '460px solid #BBDC12',
            opacity: 0.08,
            transform: `rotate(${-frame * 0.08}deg)`,

        />

            extrapolateRight: 'clamp',

          backgroundColor: '#BBDC12',
          border: '4px solid #000000',
          transform: `rotate(5deg) translateX(${interpolate(frame, [12, 52], [560, 0], {
            extrapolateRight: 'clamp',
            easing: Easing.out(Easing.cubic),
