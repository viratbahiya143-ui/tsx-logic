import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01NarrowStrip',
  durationInSeconds: 20,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01NarrowStrip: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // Count-up 0 -> 20
  const count = Math.round(interpolate(frame, [10, 55], [0, 20], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.out(Easing.cubic),
  }));

  // Camera slow zoom
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.08]);

  // Strip narrows: two dark walls close in
  const wallClose = interpolate(frame, [70, 130], [0, 1], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.inOut(Easing.cubic),
  });

  return (
    <AbsoluteFill style={{ backgroundColor: '#0BA6DF', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient + Shape Layers + Texture + Particles + Content + Vignette + Watermark */}

      {/* Dynamic gradient background */}
      <AbsoluteFill style={{
        background: `radial-gradient(circle at ${50 + Math.sin(frame * 0.02) * 10}% 40%, #2FC3F5 0%, #0BA6DF 45%, #0784B8 100%)`,
        transform: `scale(${camScale})`,
      }} />

      {/* Animated shape layers */}
      <div style={{
        position: 'absolute', width: 900, height: 900, borderRadius: '50%',
        border: '3px solid #EEEEEE', opacity: 0.06,
        left: -250 + Math.sin(frame * 0.01) * 30, top: -300,
        transform: `scale(${1 + Math.sin(frame * 0.015) * 0.05})`,
      }} />
      <div style={{
        position: 'absolute', width: 600, height: 600,
        background: '#000000', opacity: 0.05,
        right: -180, bottom: -220,
        transform: `rotate(${frame * 0.1}deg)`,
      }} />

      {/* Motion texture — horizontal flow lines (water current) */}
      {Array.from({ length: 7 }).map((_, i) => (
        <div key={`ln${i}`} style={{
          position: 'absolute', height: 2, width: 300,

      {/* Floating particles — oil drops */}

        const s = seededRandom(i + 7);
        return (
          <div key={`p${i}`} style={{
            position: 'absolute', width: 6 + s * 8, height: 6 + s * 8, borderRadius: '50%',
            background: i % 3 === 0 ? '#000000' : '#EEEEEE',
            opacity: 0.15 + s * 0.15,
            left: s * 1920,

        );
      })}

        position: 'absolute', top: 0, bottom: 0, left: 0,
