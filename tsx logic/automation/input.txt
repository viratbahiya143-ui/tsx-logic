  // Camera slow zoom
  const camScale = interpolate(frame, [0, durationInFrames], [1, 1.09], { easing: Easing.inOut(Easing.ease) });

  const tickets = [
    { label: 'ÂGE', value: 40, suffix: ' ANS', delay: 90 },

        <AbsoluteFill style={{

        {[0, 1].map((i) => (

        ))}

        {/* Motion texture — big rotating cross grid */}
        <AbsoluteFill style={{

          backgroundSize: '160px 160px',
          transform: `rotate(${Math.sin(frame * 0.005) * 2}deg) scale(1.2)`,

        {/* Floating particles */}
        {Array.from({ length: 14 }).map((_, i) => {
          const x = seededRandom(i * 9.2) * 1920;
          const y = (seededRandom(i * 3.7) * 1080 - frame * (0.4 + seededRandom(i) * 0.4) + 2160) % 1080;
          return (

        <svg width="800" height="480" viewBox="0 0 400 240" style={{ position: 'absolute', left: 90, top: 300 }}>
          {/* Ground */}
          <line x1="10" y1="220" x2="390" y2="220" stroke="#E8DCC8" strokeWidth="2.5" opacity="0.4" />
          {/* Fulcrum triangle */}
          <polygon points="150,220 190,220 170,168" fill="#F4F2F2" opacity={sp(20)} />
          {/* Beam — rotates on spring */}
          <g transform={`rotate(${leverAngle} 170 168)`}>

          <text x="60" y={156 + leverAngle * 2.4} textAnchor="middle" fontFamily="Anton, sans-serif" fontSize="22" fill="#412D15">€</text>

        <div style={{

};
